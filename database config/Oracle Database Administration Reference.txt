# Oracle Database Tables, Columns, Constraints, and Indexes Reference

## 1. List Tables

### Current Schema
===========
SELECT table_name
FROM user_tables
ORDER BY table_name;
===========

### All Accessible Schemas
===========
SELECT owner, table_name
FROM all_tables
ORDER BY owner, table_name;
===========

### DBA View
===========
SELECT owner, table_name
FROM dba_tables
ORDER BY owner, table_name;
===========

## 2. List Columns

### All Columns in Current Schema
===========
SELECT
    table_name,
    column_name,
    data_type,
    data_length,
    nullable
FROM user_tab_columns
ORDER BY table_name, column_id;
===========

### Columns of Specific Table
===========
SELECT
    column_name,
    data_type,
    data_length,
    data_precision,
    data_scale,
    nullable
FROM user_tab_columns
WHERE table_name = UPPER('TABLE_NAME')
ORDER BY column_id;
===========

### Search Column by Name
===========
SELECT
    owner,
    table_name,
    column_name,
    data_type
FROM all_tab_columns
WHERE UPPER(column_name) LIKE '%STATUS%'
ORDER BY owner, table_name;
===========

## 3. Create Constraints

### Primary Key
===========
ALTER TABLE EMPLOYEES
ADD CONSTRAINT PK_EMPLOYEES
PRIMARY KEY (EMPLOYEE_ID);
===========

### Composite Primary Key
===========
ALTER TABLE ORDER_LINES
ADD CONSTRAINT PK_ORDER_LINES
PRIMARY KEY (ORDER_ID, LINE_NUM);
===========

### Foreign Key
===========
ALTER TABLE ORDERS
ADD CONSTRAINT FK_ORDERS_CUSTOMERS
FOREIGN KEY (CUSTOMER_ID)
REFERENCES CUSTOMERS (CUSTOMER_ID);
===========

### Foreign Key with Cascade Delete
===========
ALTER TABLE ORDERS
ADD CONSTRAINT FK_ORDERS_CUSTOMERS
FOREIGN KEY (CUSTOMER_ID)
REFERENCES CUSTOMERS (CUSTOMER_ID)
ON DELETE CASCADE;
===========

### Unique Constraint
===========
ALTER TABLE USERS
ADD CONSTRAINT UK_USERS_EMAIL
UNIQUE (EMAIL);
===========

### Check Constraint
===========
ALTER TABLE ITEMS
ADD CONSTRAINT CHK_QTY
CHECK (QTY > 0);
===========

### Not Null
===========
ALTER TABLE EMPLOYEES
MODIFY EMPLOYEE_NAME NOT NULL;
===========

## 4. Remove Constraints

### Drop Primary Key
===========
ALTER TABLE EMPLOYEES
DROP PRIMARY KEY;
===========

### Drop Constraint by Name
===========
ALTER TABLE EMPLOYEES
DROP CONSTRAINT PK_EMPLOYEES;
===========

### Drop Foreign Key
===========
ALTER TABLE ORDERS
DROP CONSTRAINT FK_ORDERS_CUSTOMERS;
===========

### Drop Unique Constraint
===========
ALTER TABLE USERS
DROP CONSTRAINT UK_USERS_EMAIL;
===========

### Drop Check Constraint
===========
ALTER TABLE ITEMS
DROP CONSTRAINT CHK_QTY;
===========

### Remove Not Null
===========
ALTER TABLE EMPLOYEES
MODIFY EMPLOYEE_NAME NULL;
===========

## 5. Query Constraints

### All Constraints
===========
SELECT
    table_name,
    constraint_name,
    constraint_type,
    status
FROM user_constraints
ORDER BY table_name;
===========

### Constraint Types
===========
| Type | Description       |
| ---- | ----------------- |
| P    | Primary Key       |
| R    | Foreign Key       |
| U    | Unique            |
| C    | Check / Not Null  |
| V    | View Check Option |
| O    | View Read Only    |

### Constraints with Columns
===========
SELECT
    c.table_name,
    c.constraint_name,
    c.constraint_type,
    cc.column_name
FROM user_constraints c
JOIN user_cons_columns cc
ON c.constraint_name = cc.constraint_name
ORDER BY c.table_name;
===========

## 6. Query Foreign Keys

### Current Schema
===========
SELECT
    c.table_name,
    cc.column_name,
    r.table_name referenced_table
FROM user_constraints c
JOIN user_cons_columns cc
ON c.constraint_name = cc.constraint_name
JOIN user_constraints r
ON c.r_constraint_name = r.constraint_name
WHERE c.constraint_type = 'R';
===========

### All Schemas
===========
SELECT
    c.owner,
    c.table_name,
    cc.column_name,
    r.owner referenced_owner,
    r.table_name referenced_table
FROM all_constraints c
JOIN all_cons_columns cc
ON c.owner = cc.owner
AND c.constraint_name = cc.constraint_name
JOIN all_constraints r
ON c.r_owner = r.owner
AND c.r_constraint_name = r.constraint_name
WHERE c.constraint_type = 'R';
===========

## 7. Create Indexes

### Standard Index
===========
CREATE INDEX IDX_EMP_NAME
ON EMPLOYEES (EMPLOYEE_NAME);
===========

### Composite Index
===========
CREATE INDEX IDX_EMP_ORG_STATUS
ON EMPLOYEES (ORG_CODE, STATUS);
===========

### Unique Index
===========
CREATE UNIQUE INDEX IDX_EMAIL
ON USERS (EMAIL);
===========

### Function-Based Index
===========
CREATE INDEX IDX_UPPER_NAME
ON EMPLOYEES (UPPER(EMPLOYEE_NAME));
===========

## 8. Remove Indexes
===========
DROP INDEX IDX_EMP_NAME;
===========

## 12. Recommended Creation Order
===========
1. CREATE TABLE
2. PRIMARY KEY
3. UNIQUE CONSTRAINTS
4. FOREIGN KEYS
5. CHECK CONSTRAINTS
6. INDEXES
7. INSERT DATA

## 13. Recommended Removal Order
===========
1. FOREIGN KEYS
2. CHECK CONSTRAINTS
3. UNIQUE CONSTRAINTS
4. INDEXES
5. PRIMARY KEYS
6. TABLES


# 5. Partitioning

## Range Partition
===========
CREATE TABLE SALES (
    SALE_ID NUMBER,
    SALE_DATE DATE,
    AMOUNT NUMBER
)
PARTITION BY RANGE (SALE_DATE)
(
    PARTITION P2024 VALUES LESS THAN (DATE '2025-01-01'),
    PARTITION P2025 VALUES LESS THAN (DATE '2026-01-01'),
    PARTITION PFUTURE VALUES LESS THAN (MAXVALUE)
);
===========

## List Partition
===========
CREATE TABLE ORDERS (
    ORDER_ID NUMBER,
    ORG_CODE VARCHAR2(10)
)
PARTITION BY LIST (ORG_CODE)
(
    PARTITION P_EGYPT VALUES ('EG'),
    PARTITION P_KSA VALUES ('SA'),
    PARTITION P_OTHER VALUES (DEFAULT)
);
===========

## Hash Partition
===========
CREATE TABLE CUSTOMERS (
    CUSTOMER_ID NUMBER,
    CUSTOMER_NAME VARCHAR2(100)
)
PARTITION BY HASH (CUSTOMER_ID)
PARTITIONS 8;
===========

## Composite Partition
===========
CREATE TABLE SALES (
    SALE_ID NUMBER,
    SALE_DATE DATE,
    CUSTOMER_ID NUMBER
)
PARTITION BY RANGE (SALE_DATE)
SUBPARTITION BY HASH (CUSTOMER_ID)
SUBPARTITIONS 4
(
    PARTITION P2025 VALUES LESS THAN (DATE '2026-01-01'),
    PARTITION PFUTURE VALUES LESS THAN (MAXVALUE)
);
===========

## Add Partition
===========
ALTER TABLE SALES
ADD PARTITION P2026
VALUES LESS THAN (DATE '2027-01-01');
===========

## Drop Partition
===========
ALTER TABLE SALES
DROP PARTITION P2024;
===========

## Truncate Partition
===========
ALTER TABLE SALES
TRUNCATE PARTITION P2024;
===========

## Move Partition
===========
ALTER TABLE SALES
MOVE PARTITION P2025;
===========

## Query Partitions
===========
SELECT
    table_name,
    partition_name,
    high_value
FROM user_tab_partitions;
===========

## Local Index
===========
CREATE INDEX IDX_SALES_DATE
ON SALES (SALE_DATE)
LOCAL;
===========

## Global Index
===========
CREATE INDEX IDX_SALES_ID
ON SALES (SALE_ID)
GLOBAL;
===========

# 7. Recommended Creation Order
===========
1. Create Tables
2. Create Primary Keys
3. Create Unique Constraints
4. Create Foreign Keys
5. Create Check Constraints
6. Create Indexes
7. Load Data

# 8. Recommended Removal Order
===========
1. Drop Foreign Keys
2. Drop Check Constraints
3. Drop Unique Constraints
4. Drop Indexes
5. Drop Primary Keys
6. Drop Tables
