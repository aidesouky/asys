SELECT
  TAB.OWNER,
  TAB.TABLE_NAME,
  COL.COLUMN_ID,
  COL.COLUMN_NAME,
  COL.DATA_TYPE,
  COL.DATA_LENGTH,
  COL.DATA_PRECISION,
  COL.DATA_SCALE,
  COL.NULLABLE,
  OBJ.CREATED,
  OBJ.LAST_DDL_TIME
FROM
  ALL_TABLES TAB,
  ALL_TAB_COLUMNS COL,
  ALL_OBJECTS OBJ
WHERE
  1 = 1
  AND   TAB.TABLE_NAME = COL.TABLE_NAME
  AND   TAB.TABLE_NAME = OBJ.OBJECT_NAME
  AND   TAB.OWNER = UPPER('SCHEMA_NAME')
  AND   OBJ.OBJECT_TYPE = 'TABLE'
ORDER BY
  TAB.TABLE_NAME,
  COL.COLUMN_ID;

SELECT
  FK.OWNER,
  FK.TABLE_NAME AS FK_TABLE,
  FK_COL.COLUMN_NAME AS FK_COLUMN,
  PK.OWNER AS PK_OWNER,
  PK.TABLE_NAME AS PK_TABLE,
  PK_COL.COLUMN_NAME AS PK_COLUMN,
  FK.CONSTRAINT_NAME AS FK_NAME
FROM
  ALL_CONSTRAINTS FK,
  ALL_CONS_COLUMNS FK_COL,
  ALL_CONSTRAINTS PK,
  ALL_CONS_COLUMNS PK_COL
WHERE
  FK.OWNER = FK_COL.OWNER
  AND   FK.CONSTRAINT_NAME = FK_COL.CONSTRAINT_NAME
  AND   FK.R_OWNER = PK.OWNER
  AND   FK.R_CONSTRAINT_NAME = PK.CONSTRAINT_NAME
  AND   PK.OWNER = PK_COL.OWNER
  AND   PK.CONSTRAINT_NAME = PK_COL.CONSTRAINT_NAME
  AND   FK_COL.POSITION = PK_COL.POSITION
  AND   FK.CONSTRAINT_TYPE = 'R'
  AND   FK.OWNER = UPPER('SCHEMA_NAME')
ORDER BY
  FK.OWNER,
  FK.TABLE_NAME;
  /
  SELECT
  I.OWNER,
  I.TABLE_NAME,
  I.INDEX_NAME,
  I.INDEX_TYPE,
  I.UNIQUENESS,
  C.COLUMN_POSITION,
  C.COLUMN_NAME
FROM
  ALL_INDEXES I,
  ALL_IND_COLUMNS C
WHERE
  I.OWNER = C.INDEX_OWNER
  AND   I.INDEX_NAME = C.INDEX_NAME
  AND   I.OWNER = UPPER('SCHEMA_NAME')
ORDER BY
  I.TABLE_NAME,
  I.INDEX_NAME,
  C.COLUMN_POSITION;
/
  SELECT
  FK.OWNER,
  FK.TABLE_NAME AS FK_TABLE,
  FK_COL.COLUMN_NAME AS FK_COLUMN,
  PK.OWNER AS PK_OWNER,
  PK.TABLE_NAME AS PK_TABLE,
  PK_COL.COLUMN_NAME AS PK_COLUMN,
  FK.CONSTRAINT_NAME AS FK_NAME
FROM
  ALL_CONSTRAINTS FK,
  ALL_CONS_COLUMNS FK_COL,
  ALL_CONSTRAINTS PK,
  ALL_CONS_COLUMNS PK_COL
WHERE
  FK.OWNER = FK_COL.OWNER
  AND   FK.CONSTRAINT_NAME = FK_COL.CONSTRAINT_NAME
  AND   FK.R_OWNER = PK.OWNER
  AND   FK.R_CONSTRAINT_NAME = PK.CONSTRAINT_NAME
  AND   PK.OWNER = PK_COL.OWNER
  AND   PK.CONSTRAINT_NAME = PK_COL.CONSTRAINT_NAME
  AND   FK_COL.POSITION = PK_COL.POSITION
  AND   FK.CONSTRAINT_TYPE = 'R'
  AND   FK.OWNER = UPPER('SCHEMA_NAME')
  AND   FK_COL.COLUMN_NAME NOT IN (
    SELECT
      C.COLUMN_NAME
    FROM
      ALL_INDEXES I,
      ALL_IND_COLUMNS C
    WHERE
      I.OWNER = C.INDEX_OWNER
      AND   I.INDEX_NAME = C.INDEX_NAME
      AND   I.OWNER = UPPER('SCHEMA_NAME')
  )
ORDER BY
  FK.OWNER,
  FK.TABLE_NAME;
