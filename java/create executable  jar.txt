Create Executable jar file
==========================
1.create java class
# touch MyClass.java
-- for class in package "com.myapp"
# touch com/myapp/MyClass.java

2.write code into java file
public class MyClass {
    public static void main(String[] args) {
        System.out.println("Hello, world!");
    }
}
-- for class in package "com.myapp"
package com.myapp;
public class MyClass {
    public static void main(String[] args) {
        System.out.println("Hello, world!");
    }
}


3.compile java file to generate MyClass.class
# javac MyClass.java
# javac -cp ".:lib/*" -d src com/myapp/MyClass.java

4.create jar with class file

# javac -cp ".:lib/*" -d src com/myapp/MyClass.java
# jar cfm myapp.jar manifest.txt -C src .

# jar cfe myapp.jar com.myapp.MyClass com/myapp/
# javac -d src com/myapp/*.java
# jar cfe myapp.jar com.myapp.MyClass -C src .
