JAR Command
===========
make "files" directory as jar file
# jar cf file.jar ./files

make "files" directory as jar file without create "MANIFEST.MF" file
# jar cfM file.jar ./files

list all files and directories in "file.jar"
# jar tf file.jar

extract all files and directories in "file.jar"
# jar xf file.jar

extract "file.txt" from file.jar
# jar xf file.jar files/file.txt

insert "file.txt" into "files" directory in "file.jar"
# jar uf file.jar files/file.txt

update "MANIFEST.MF" file
# jar ufm file.jar META-INF/MANIFEST.MF
# jar -u -f file.jar -m META-INF/MANIFEST.MF

create jar file with main class "com.myapp.MainClass" and MainClass file
# jar cfe file.jar com.myapp.MainClass ./com/myapp/MainClass.class
# jar c -f file.jar -e com.myapp.MainClass ./com/myapp/MainClass.class

change Main Class in "MANIFEST.MF" file to "com.myapp.MainClass"
# jar ufe file.jar com.myapp.MainClass
# jar u -f file.jar -e com.myapp.MainClass

change Main Class in "MANIFEST.MF" file to "com.myapp.MainClass" and insert MainClass file
# jar ufe file.jar com.myapp.MainClass ./com/myapp/MainClass.class
# jar u -f file.jar -e com.myapp.MainClass ./com/myapp/MainClass.class
