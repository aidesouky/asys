            String password = "ETACDB";
            Connection con = DriverManager.getConnection(url, user, password);
            CallableStatement cs = con.prepareCall("{ ? = call mypkg.myfunc(?) }");
			//CallableStatement cs = con.prepareCall("BEGIN ? := mypkg.myfunc(?); END;");
