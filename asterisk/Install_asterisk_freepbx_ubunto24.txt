[1] Update system packages
# sudo apt update 
# sudo apt upgrade -y

[2] Fix PATH (important for fwconsole & asterisk):
# echo 'export PATH=$PATH:/sbin:/usr/sbin' >> ~/.bashrc
# source ~/.bashrc

[3] Install Core Build Dependencies: 
# sudo apt install -y ssh wget curl git unzip gnupg2 pkg-config build-essential subversion libicu-dev libnewt-dev libncurses5-dev libjansson-dev libxml2-dev libssl-dev uuid-dev sox libedit-dev sqlite3 libsqlite3-dev
# sudo apt install -y incron --no-install-recommends

[4] Download, Build and Install Asterisk
# cd /usr/src
# sudo wget http://downloads.asterisk.org/pub/telephony/asterisk/asterisk-18-current.tar.gz
# sudo tar -zxvf asterisk-18-current.tar.gz
# cd asterisk-18.*
# sudo contrib/scripts/get_mp3_source.sh
# sudo contrib/scripts/install_prereq install
# sudo ./configure
# sudo make menuselect
# sudo menuselect/menuselect --enable app_macro --enable format_mp3 --disable BUILD_NATIVE menuselect.makeopts
# sudo make -j2
# sudo make install
# sudo make samples
# sudo make config
# sudo ldconfig

[5] Create asterisk (user, group),set permissions and allow asterisk (user, group)
# sudo groupadd asterisk
# sudo useradd -r -d /var/lib/asterisk -g asterisk asterisk
# sudo usermod -aG audio,dialout,video,plugdev,cdrom asterisk
# sudo chown -R asterisk:asterisk /etc/asterisk
# sudo chown -R asterisk:asterisk /var/{lib,log,spool}/asterisk
# sudo chown -R asterisk:asterisk /usr/lib/asterisk

# sudo sed -i 's|;runuser = asterisk|runuser = asterisk|g' /etc/asterisk/asterisk.conf
# sudo sed -i 's|;rungroup = asterisk|rungroup = asterisk|g' /etc/asterisk/asterisk.conf
# sudo grep -E ^"runuser|rungroup" /etc/asterisk/asterisk.conf
# sudo sed -i 's|#AST_USER="asterisk"|AST_USER="asterisk"|g' /etc/default/asterisk
# sudo sed -i 's|#AST_GROUP="asterisk"|AST_GROUP="asterisk"|g' /etc/default/asterisk
# sudo grep -E ^"AST_USER|AST_GROUP" /etc/default/asterisk

# sudo nano /etc/asterisk/asterisk.conf
