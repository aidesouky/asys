# sudo nano /etc/asterisk/extensions_custom.conf

[check-balance]
exten => s,1,Answer()

 same => n,Playback(custom/enter_account_sound)
 same => n,Read(ACCID,,6)

 same => n,Playback(custom/enter_pin_sound)
 same => n,Read(PIN,,4)

 ; DEBUG: show what was entered
 same => n,NoOp(ACCID=${ACCID})
 same => n,NoOp(PIN=${PIN})
 
 ; optional PIN check
 same => n,GotoIf($["${DB(customer/${ACCID})}"="${PIN}"]?valid:invalid)
 
 same => n(valid),Set(BALANCE=${IF($["${DB(balance/${ACCID})}"=""]?0:${DB(balance/${ACCID})})})
 same => n,NoOp(VALUE=${BALANCE})

 same => n,Playback(custom/your_balance_sound)
 same => n,SayNumber(${BALANCE})
 same => n,Playback(custom/pound_sound)

 same => n,Hangup()

 ;same => n,Goto(app-announcement,2,1)

==CREATE AGI PHP SCRIPT
# sudo nano /var/lib/asterisk/agi-bin/get_balance.php
------------------------
#!/usr/bin/php -q
<?php
require_once('phpagi.php');
$agi = new AGI();
/* Get arguments from dialplan */
$code = $argv[1];
$pin  = $argv[2];
/* Database connection */
$db = new mysqli("127.0.0.1", "root", "181084", "balance_db");
if ($db->connect_error) {
    $agi->verbose("DB Connection Failed", 1);
    $agi->set_variable("BALANCE", "0");
    exit;
}
/* Validate customer */
$sql = "SELECT balance FROM customers WHERE code=? AND pin=?";
$stmt = $db->prepare($sql);
$stmt->bind_param("ss", $code, $pin);
$stmt->execute();
$stmt->bind_result($balance);
if ($stmt->fetch()) {
    $agi->verbose("Balance found: $balance", 1);
    $agi->set_variable("BALANCE", $balance);
} else {
    $agi->verbose("Invalid credentials", 1);
    $agi->set_variable("BALANCE", "INVALID");
}
$stmt->close();
$db->close();
?>
-------------------------------------------

==CREATE AGI PHP SCRIPT
# sudo nano /var/lib/asterisk/agi-bin/update_pin.php
------------------------
#!/usr/bin/php -q
<?php
require_once('phpagi.php');
$agi = new AGI();

$code     = $argv[1];
$old_pin  = $argv[2];
$new_pin  = $argv[3];

$db = new mysqli("127.0.0.1", "root", "181084", "balance_db");

if ($db->connect_error) {
    $agi->verbose("DB Connection Failed", 1);
    $agi->set_variable("PIN_STATUS", "ERROR");
    exit;
}

/* verify old pin */
$sql = "SELECT id FROM customers WHERE code=? AND pin=?";
$stmt = $db->prepare($sql);
$stmt->bind_param("ss", $code, $old_pin);
$stmt->execute();
$stmt->store_result();

if ($stmt->num_rows == 0) {
    $agi->verbose("Invalid old PIN", 1);
    $agi->set_variable("PIN_STATUS", "INVALID");
    exit;
}
$stmt->close();

/* update pin */
$update = $db->prepare("UPDATE customers SET pin=? WHERE code=?");
$update->bind_param("ss", $new_pin, $code);

if ($update->execute()) {
    $agi->verbose("PIN Updated Successfully", 1);
    $agi->set_variable("PIN_STATUS", "OK");
} else {
    $agi->set_variable("PIN_STATUS", "ERROR");
}

$update->close();
$db->close();
?>

-------------------------------------------

=CREATE DIALPLAN TO CALL AGI PHP SCRIPT
# sudo nano /etc/asterisk/extensions_custom.conf
----------------------------------------------------
[change-pin-agi-ar]
exten => s,1,Answer()

 same => n,Playback(custom/enter_account_sound_ar)
 same => n,Read(ACCID,,6)
 same => n,Playback(custom/enter_pin_sound_ar)
 same => n,Read(PIN,,6)
 same => n,Playback(custom/new_pin_sound_ar)
 same => n,Read(NPIN,,6)
 same => n,Playback(custom/retry_pin_sound_ar)
 same => n,Read(RPIN,,6)
 
 ; check new pin match
 same => n,GotoIf($["${NPIN}"="${RPIN}"]?pin_ok:pin_no)
 
 ; Call PHP AGI
 same => n(pin_ok),AGI(update_pin.php,${ACCID},${PIN},${NPIN})

 ; Check result
 same => n,GotoIf($["${PIN_STATUS}"="OK"]?success:failed)
 
 same => n(success),Playback(custom/valid_operation_sound_ar)
 same => n,Playback(custom/thank_you_sound_ar)
 same => n,Hangup()
 
 same => n(pin_no),Playback(custom/pin_match_sound_ar)
 same => n,Playback(custom/invalid_operation_sound_ar)
 same => n,Playback(custom/thank_you_sound_ar)
 same => n,Hangup()
 
 same => n(failed),Playback(custom/invalid_operation_sound_ar)
 same => n,Playback(custom/thank_you_sound_ar)
 same => n,Hangup()
 --------------------------------------------------------
